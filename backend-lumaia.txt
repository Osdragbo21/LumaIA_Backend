// Ruta: src/citas/entities/cita.entity.ts
import { ObjectType, Field, ID, registerEnumType } from '@nestjs/graphql';
import { Prop, Schema, SchemaFactory } from '@nestjs/mongoose';
import { Document, Types } from 'mongoose';

// Definición de estados del ciclo de vida de la cita
export enum EnumEstadoCita {
  PROGRAMADA = 'Programada',
  COMPLETADA = 'Completada',
  CANCELADA = 'Cancelada',
}

registerEnumType(EnumEstadoCita, { name: 'EnumEstadoCita' });

@ObjectType()
@Schema({ timestamps: true })
export class Cita extends Document {
  @Field(() => ID)
  _id: Types.ObjectId;

  @Field(() => ID)
  @Prop({ type: Types.ObjectId, ref: 'Usuario', required: true })
  usuario_id: Types.ObjectId;

  @Field()
  @Prop({ required: true })
  titulo_evento: string;

  @Field()
  @Prop({ required: true })
  fecha_hora: Date;

  @Field({ nullable: true })
  @Prop()
  ubicacion?: string;

  @Field(() => EnumEstadoCita)
  @Prop({ required: true, enum: EnumEstadoCita, default: EnumEstadoCita.PROGRAMADA })
  estado: EnumEstadoCita;

  // Sello de tiempo para ejecución del índice TTL (retención de 30 días)
  @Field(() => Date, { nullable: true })
  @Prop({ type: Date })
  fecha_eliminacion?: Date;
}

export const CitaSchema = SchemaFactory.createForClass(Cita);

// Índice TTL para destruir el documento 30 días después de ser marcado como eliminado
CitaSchema.index({ fecha_eliminacion: 1 }, { expireAfterSeconds: 2592000 });

import { Resolver, Mutation, Args, ID, ObjectType, Field } from '@nestjs/graphql';
import { AsistenteService } from './asistente.service';

// 1. Definimos la nueva estructura de respuesta para GraphQL (JSON Estructurado)
@ObjectType()
export class AsistenteResponse {
  @Field()
  respuesta_texto: string;

  @Field()
  intencion: string;
}

@Resolver()
export class AsistenteResolver {
  constructor(private readonly asistenteService: AsistenteService) {}

  @Mutation(() => AsistenteResponse, { name: 'consultarAsistente' })
  async consultarAsistente(
    @Args('usuario_id', { type: () => ID }) usuario_id: string,
    @Args('pregunta_id') pregunta_id: string,
    @Args('modelo', { type: () => String, nullable: true }) modelo?: string,
  ): Promise<AsistenteResponse> {
    // Delegamos al servicio y retornamos el objeto estructurado
    return this.asistenteService.procesarPregunta(usuario_id, pregunta_id, modelo);
  }
}

import { Injectable } from '@nestjs/common';
import { InjectModel } from '@nestjs/mongoose';
import { Model } from 'mongoose';
import { JwtService } from '@nestjs/jwt';
import { GraphQLError } from 'graphql';
import * as bcrypt from 'bcrypt';
import { Usuario } from '../usuarios/entities/usuario.entity';
import { LoginResponse } from './dto/login-response.dto';

@Injectable()
export class AuthService {
  constructor(
    @InjectModel(Usuario.name) private usuarioModel: Model<Usuario>,
    private jwtService: JwtService,
  ) {}

  async validarUsuario(correo: string, password_plana: string): Promise<LoginResponse> {
    // 1. Buscar usuario por correo en MongoDB Atlas
    const usuario = await this.usuarioModel.findOne({ correo }).exec();
    if (!usuario) {
      throw new GraphQLError('Correo o contraseña incorrectos.', { extensions: { code: 'UNAUTHENTICATED' } });
    }

    // 2. Comparar la contraseña plana contra el Hash seguro (RNF-06)
    const passwordValida = await bcrypt.compare(password_plana, usuario.password_hash);
    if (!passwordValida) {
      throw new GraphQLError('Correo o contraseña incorrectos.', { extensions: { code: 'UNAUTHENTICATED' } });
    }

    // 3. Validar borrado lógico
    if (!usuario.estado_activo) {
      throw new GraphQLError('Esta cuenta se encuentra desactivada.', { extensions: { code: 'FORBIDDEN' } });
    }

    // 4. Firmar el Token JWT (RF-03.1)
    const payload = { sub: usuario._id, rol: usuario.rol };
    const access_token = await this.jwtService.signAsync(payload);

    return {
      access_token,
      usuario_id: usuario._id.toString(),
      rol: usuario.rol,
    };
  }
}

import { Injectable } from '@nestjs/common';
import { InjectModel } from '@nestjs/mongoose';
import { Model, Types } from 'mongoose';
import { GraphQLError } from 'graphql';
import { RegistroToma, Enum_EstadoToma } from './entities/registro-toma.entity';
import { Medicamento } from '../medicamentos/entities/medicamento.entity';


@Injectable()
export class RegistrosTomaService {
  constructor(
    @InjectModel(RegistroToma.name) private registroTomaModel: Model<RegistroToma>,
    @InjectModel(Medicamento.name) private medicamentoModel: Model<Medicamento>, // Inyectamos Medicamento
  ) {}

  async confirmarToma(medicamento_id: string): Promise<RegistroToma> {
    try {
      const nuevaToma = new this.registroTomaModel({
        medicamento_id: new Types.ObjectId(medicamento_id),
        fecha_programada: new Date(), 
        estado_toma: Enum_EstadoToma.CONFIRMADA,
        fecha_confirmacion: new Date(),
      });
      
      return await nuevaToma.save();
    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : 'Error desconocido';
      throw new GraphQLError(`Error al registrar la toma: ${errorMessage}`, {
        extensions: { code: 'INTERNAL_SERVER_ERROR' },
      });
    }
  }

  // NUEVO: Obtener las tomas confirmadas del día actual
  async obtenerTomasHoy(usuario_id: string): Promise<RegistroToma[]> {
    try {
      const usuarioObjectId = new Types.ObjectId(usuario_id);

      // 1. Obtener los IDs de todos los medicamentos de este usuario
      const medicamentosUsuario = await this.medicamentoModel.find({ usuario_id: usuarioObjectId }).exec();
      const idsMedicamentos = medicamentosUsuario.map(med => med._id);

      // 2. Definir el rango del día actual (00:00:00 a 23:59:59)
      const inicioDia = new Date();
      inicioDia.setHours(0, 0, 0, 0);
      
      const finDia = new Date();
      finDia.setHours(23, 59, 59, 999);

      // 3. Buscar los registros de hoy que estén confirmados
      return await this.registroTomaModel.find({
        medicamento_id: { $in: idsMedicamentos },
        estado_toma: Enum_EstadoToma.CONFIRMADA,
        fecha_confirmacion: { $gte: inicioDia, $lte: finDia }
      }).exec();

    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : 'Error desconocido';
      throw new GraphQLError(`Error al consultar las tomas de hoy: ${errorMessage}`, {
        extensions: { code: 'INTERNAL_SERVER_ERROR' },
      });
    }
  }
}

// [BACKEND] Ruta: src/asistente/asistente.service.ts

import { Injectable } from '@nestjs/common';
import { HttpService } from '@nestjs/axios';
import { InjectModel } from '@nestjs/mongoose';
import { Model, Types } from 'mongoose';
import { firstValueFrom } from 'rxjs';
import { GraphQLError } from 'graphql';
import { Medicamento } from '../medicamentos/entities/medicamento.entity';
import { LogInteraccion } from '../logs-interaccion/entities/log-interaccion.entity';
import { Cita } from '../citas/entities/cita.entity';
import { RegistroToma, Enum_EstadoToma } from '../registros-toma/entities/registro-toma.entity';

interface IAsistenteResponse {
  respuesta_texto: string;
  intencion: string;
}

@Injectable()
export class AsistenteService {
  constructor(
    private readonly httpService: HttpService,
    @InjectModel(Medicamento.name) private medicamentoModel: Model<Medicamento>,
    @InjectModel(LogInteraccion.name) private logInteraccionModel: Model<LogInteraccion>,
    @InjectModel(Cita.name) private citaModel: Model<Cita>,
    @InjectModel(RegistroToma.name) private registroTomaModel: Model<RegistroToma>,
  ) {}

  private clasificarIntencionRapida(pregunta: string): string {
    const p = pregunta.toLowerCase();
    if (p.includes('agregar') || p.includes('nuevo') || p.includes('crear') || p.includes('registrar')) {
      if (p.includes('medicina') || p.includes('pastilla') || p.includes('tratamiento')) return 'ACCION_NUEVO_MEDICAMENTO';
      if (p.includes('cita') || p.includes('doctor') || p.includes('consulta')) return 'ACCION_NUEVA_CITA';
      if (p.includes('contacto') || p.includes('familiar') || p.includes('apoyo')) return 'ACCION_NUEVO_CONTACTO';
    }
    return 'CONSULTA_GENERAL';
  }

  private parsearRespuestaJSON(respuestaRaw: string, intencionRespaldo: string): IAsistenteResponse {
    try {
      const cleanJson = respuestaRaw.replace(/```json/g, '').replace(/```/g, '').trim();
      const parsed = JSON.parse(cleanJson);
      return {
        respuesta_texto: parsed.respuesta_texto ? parsed.respuesta_texto.replace(/"/g, '') : "Lo siento, no pude procesar eso.",
        intencion: parsed.intencion || intencionRespaldo
      };
    } catch (error) {
      return {
        respuesta_texto: respuestaRaw.replace(/"/g, ''),
        intencion: intencionRespaldo
      };
    }
  }

  async procesarPregunta(usuario_id: string, pregunta_id: string, modelo: string = 'Luma Cloud'): Promise<IAsistenteResponse> {
    try {
      const objectId = new Types.ObjectId(usuario_id);

      // --- A. EXTRACCIÓN DE CONTEXTO ---
      const medicamentos = await this.medicamentoModel.find({ usuario_id: objectId, fecha_eliminacion: { $exists: false } }).exec();
      const inicioDia = new Date(); inicioDia.setHours(0, 0, 0, 0);
      const finDia = new Date(); finDia.setHours(23, 59, 59, 999);

      const citasHoyDocs = await this.citaModel.find({ usuario_id: objectId, fecha_eliminacion: { $exists: false }, fecha_hora: { $gte: inicioDia, $lte: finDia } }).exec();
      const tomasHoyDocs = await this.registroTomaModel.find({
        medicamento_id: { $in: medicamentos.map(m => m._id) },
        estado_toma: Enum_EstadoToma.CONFIRMADA,
        fecha_confirmacion: { $gte: inicioDia, $lte: finDia }
      }).exec();

      // CORRECCIÓN CRÍTICA: Aseguramos que los IDs se comparen estrictamente como Strings
      const idsTomadosHoy = tomasHoyDocs.map(t => t.medicamento_id.toString());
      const medicamentos_activos = medicamentos.map(med => ({ id: med._id.toString(), nombre_farmaco: med.nombre_farmaco, horarios_especificos: med.horarios_especificos }));
      
      const citas_hoy = citasHoyDocs.map(cita => ({ titulo_evento: cita.titulo_evento, fecha_hora: cita.fecha_hora.toISOString(), ubicacion: cita.ubicacion || 'Sin ubicación' }));
      const medsPendientes = medicamentos_activos.filter(m => !idsTomadosHoy.includes(m.id));
      const medsTomados = medicamentos_activos.filter(m => idsTomadosHoy.includes(m.id));

      // --- B. CONSTRUCCIÓN DEL SYSTEM PROMPT CENTRALIZADO ---
      const systemPrompt = `
        Eres Luma, el asistente personal del usuario. 
        REGLAS ESTRICTAS:
        1. Háblale SIEMPRE de "tú" de forma muy cálida y humana.
        2. Los nombres en la lista de "Medicamentos" son NOMBRES COMERCIALES DE MEDICINAS. NUNCA los interpretes como estudios de laboratorio, procedimientos ni pruebas médicas, sin importar lo raros que suenen.
        3. Sé extremadamente breve y nunca uses comillas para envolver tu respuesta.
        4. Responde ÚNICAMENTE entregando un objeto JSON válido con este formato exacto: {"respuesta_texto": "Tu respuesta", "intencion": "TIPO_DE_INTENCION"}
        5. Intenciones válidas: CONSULTA_GENERAL, ACCION_NUEVO_MEDICAMENTO, ACCION_NUEVA_CITA.

        CONTEXTO HOY:
        - Citas: ${citas_hoy.map(c => c.titulo_evento).join(', ') || 'Ninguna'}.
        - Medicamentos YA TOMADOS: ${medsTomados.map(m => m.nombre_farmaco).join(', ') || 'Ninguno'}.
        - Medicamentos PENDIENTES por tomar: ${medsPendientes.map(m => m.nombre_farmaco).join(', ') || 'Ninguno'}.

        EL USUARIO DICE: "${pregunta_id}"
      `;

      const intencionRapida = this.clasificarIntencionRapida(pregunta_id);
      let resultadoFinal: IAsistenteResponse;

      // --- C. ENRUTADOR DE MODELOS IA (Una sola pasada) ---
      if (modelo === 'Llama 3.2 Local') {
        const ollamaPayload = {
          model: "llama3.2",
          prompt: systemPrompt,
          format: "json", 
          stream: false
        };

        const { data } = await firstValueFrom(
          this.httpService.post('http://127.0.0.1:11434/api/generate', ollamaPayload)
        );
        resultadoFinal = this.parsearRespuestaJSON(data.response, intencionRapida);
      } else {
        const { data } = await firstValueFrom(
          this.httpService.post('http://localhost:8000/evaluar', {
            pregunta_id,
            medicamentos_activos,
            citas: citas_hoy,
            tomas_hoy_ids: idsTomadosHoy,
            prompt_forzado: systemPrompt 
          })
        );
        if (data.respuesta_texto && data.intencion) {
          resultadoFinal = { respuesta_texto: data.respuesta_texto, intencion: data.intencion };
        } else if (data.respuesta) {
          resultadoFinal = this.parsearRespuestaJSON(data.respuesta, intencionRapida);
        } else {
          resultadoFinal = { respuesta_texto: "Lo siento, hubo un problema al generar la respuesta.", intencion: 'CONSULTA_GENERAL' };
        }
      }

      // --- D. AUDITORÍA ---
      const nuevoLog = new this.logInteraccionModel({
        usuario_id: objectId,
        premisa_entrada: pregunta_id,
        respuesta_generada: resultadoFinal.respuesta_texto,
        bloqueo_medico: false,
        fecha_interaccion: new Date(),
      });
      await nuevoLog.save();

      return resultadoFinal;
    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : 'Error desconocido';
      throw new GraphQLError(`Fallo en el motor de inferencia: ${errorMessage}`, {
        extensions: { code: 'INTERNAL_SERVER_ERROR' },
      });
    }
  }
}

import { Injectable } from '@nestjs/common';
import { InjectModel } from '@nestjs/mongoose';
import { Model, Types } from 'mongoose';
import { GraphQLError } from 'graphql';
import * as bcrypt from 'bcrypt';
import { Usuario } from './entities/usuario.entity';
import { Medicamento } from '../medicamentos/entities/medicamento.entity';
import { Cita } from '../citas/entities/cita.entity';
import { RegistroToma, Enum_EstadoToma } from '../registros-toma/entities/registro-toma.entity';

@Injectable()
export class UsuariosService {
  constructor(
    @InjectModel(Usuario.name) private usuarioModel: Model<Usuario>,
    @InjectModel(Medicamento.name) private medicamentoModel: Model<Medicamento>,
    @InjectModel(Cita.name) private citaModel: Model<Cita>,
    @InjectModel(RegistroToma.name) private registroTomaModel: Model<RegistroToma>,
  ) {}

  // ... (MANTÉN TUS MÉTODOS ACTUALES DE REGISTRO, BUSCAR POR ID, AGREGAR CONTACTO, ETC.) ...

  // NUEVO: Vincular por correo
  async vincularAdultoMayor(cuidador_id: string, correo_adulto: string): Promise<Usuario> {
    try {
      const cuidadorObjectId = new Types.ObjectId(cuidador_id);
      
      const adulto = await this.usuarioModel.findOne({ correo: correo_adulto, rol: 'Adulto Mayor' }).exec();
      if (!adulto) throw new Error('No se encontró un Adulto Mayor con ese correo.');

      // Agregamos al cuidador
      await this.usuarioModel.findByIdAndUpdate(cuidadorObjectId, {
        $addToSet: { adultos_vinculados: adulto._id }
      }).exec();

      // Actualizamos al adulto
      adulto.cuidador_vinculado_id = cuidadorObjectId;
      return await adulto.save();
    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : 'Error desconocido';
      throw new GraphQLError(`Fallo al vincular: ${errorMessage}`);
    }
  }

  // NUEVO: Resumen Completo para el Cuidador
  async obtenerAdultosVinculados(cuidador_id: string): Promise<any[]> {
    try {
      const cuidador = await this.usuarioModel.findById(new Types.ObjectId(cuidador_id)).exec();
      if (!cuidador || !cuidador.adultos_vinculados || cuidador.adultos_vinculados.length === 0) return [];

      const adultos = await this.usuarioModel.find({ _id: { $in: cuidador.adultos_vinculados } }).exec();
      
      const inicioDia = new Date(); inicioDia.setHours(0, 0, 0, 0);
      const finDia = new Date(); finDia.setHours(23, 59, 59, 999);

      const resumenAdultos = await Promise.all(adultos.map(async (adulto) => {
        const medicamentos = await this.medicamentoModel.find({ usuario_id: adulto._id, fecha_eliminacion: { $exists: false } }).exec();
        const citas_hoy = await this.citaModel.find({ usuario_id: adulto._id, fecha_eliminacion: { $exists: false }, fecha_hora: { $gte: inicioDia, $lte: finDia } }).exec();
        const tomas_hoy = await this.registroTomaModel.find({ medicamento_id: { $in: medicamentos.map(m => m._id) }, estado_toma: Enum_EstadoToma.CONFIRMADA, fecha_confirmacion: { $gte: inicioDia, $lte: finDia } }).exec();
        
        return {
          adulto,
          medicamentos,
          citas_hoy,
          tomas_hoy
        };
      }));

      return resumenAdultos;
    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : 'Error desconocido';
      throw new GraphQLError(`Error al consultar vinculados: ${errorMessage}`);
    }
  }
}

// [BACKEND] Ruta: src/medicamentos/medicamentos.service.ts

import { Injectable } from '@nestjs/common';
import { InjectModel } from '@nestjs/mongoose';
import { Model, Types } from 'mongoose';
import { GraphQLError } from 'graphql';
import { Medicamento } from './entities/medicamento.entity';
import { CreateMedicamentoInput } from './dto/create-medicamento.input';

@Injectable()
export class MedicamentosService {
  constructor(
    @InjectModel(Medicamento.name) private medicamentoModel: Model<Medicamento>,
  ) {}

  async create(input: CreateMedicamentoInput): Promise<Medicamento> {
    try {
      const usuarioObjectId = new Types.ObjectId(input.usuario_id);
      
      // 1. Clonamos el arreglo original para manipularlo
      let horariosCalculados = input.horarios_especificos;

      // 2. Algoritmo de Cálculo de Horarios
      if (
        input.horarios_especificos && 
        input.horarios_especificos.length > 0 && 
        input.frecuencia_horas && 
        input.frecuencia_horas > 0
      ) {
        // Extraemos la primera toma (Ej. "08:30")
        const horaInicialStr = input.horarios_especificos[0];
        const partesHora = horaInicialStr.split(':');
        
        if (partesHora.length === 2) {
          const horaInicial = parseInt(partesHora[0], 10);
          const minutos = parseInt(partesHora[1], 10);

          // Límite de tomas al día (Ej. 24 / 8 = 3 tomas)
          const limiteTomasDia = Math.floor(24 / input.frecuencia_horas);
          
          horariosCalculados = []; // Vaciamos para llenarlo con la secuencia completa

          // Generamos la secuencia de horarios
          for (let i = 0; i < limiteTomasDia; i++) {
            // Calculamos la hora sumando la frecuencia y usando módulo 24 para saltos de medianoche
            const horaCalculada = (horaInicial + (i * input.frecuencia_horas)) % 24;
            
            // Formateo seguro a string de dos dígitos (HH:mm)
            const hh = horaCalculada.toString().padStart(2, '0');
            const mm = minutos.toString().padStart(2, '0');
            
            horariosCalculados.push(`${hh}:${mm}`);
          }
        }
      }

      // 3. Instanciar documento con el arreglo calculado y guardar
      const nuevoMedicamento = new this.medicamentoModel({
        ...input,
        usuario_id: usuarioObjectId,
        horarios_especificos: horariosCalculados, // Reemplazamos con el nuevo arreglo
      });

      return await nuevoMedicamento.save();
    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : 'Error desconocido';
      throw new GraphQLError(`Fallo al guardar el medicamento: ${errorMessage}`, {
        extensions: { code: 'BAD_USER_INPUT' },
      });
    }
  }

  // MÉTODO DE LECTURA INTACTO
  async findAllByUserId(usuario_id: string): Promise<Medicamento[]> {
    try {
      const usuarioObjectId = new Types.ObjectId(usuario_id);
      // Busca todos los medicamentos de este usuario que no hayan sido marcados para borrado lógico (TTL)
      return await this.medicamentoModel.find({ 
        usuario_id: usuarioObjectId,
        fecha_eliminacion: { $exists: false } 
      }).exec();
    } catch (error) {
      const errorMessage = error instanceof Error ? error.message : 'Error desconocido';
      throw new GraphQLError(`Error al consultar medicamentos: ${errorMessage}`, {
        extensions: { code: 'INTERNAL_SERVER_ERROR' },
      });
    }
  }
}
